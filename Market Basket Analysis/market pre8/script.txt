-- Creación de la Base de Datos para el Segundo Caso: Análisis de Cestas de Mercado
CREATE DATABASE MarketBasketAnalysis;
USE MarketBasketAnalysis;

-- Tabla de Transacciones
CREATE TABLE Transactions (
    transaction_id INT PRIMARY KEY,
    transaction_date DATE NOT NULL
);

-- Tabla de Artículos Comprados en cada Transacción
CREATE TABLE Transaction_Items (
    transaction_item_id INT PRIMARY KEY,
    transaction_id INT,
    item_name VARCHAR(50),
    FOREIGN KEY (transaction_id) REFERENCES Transactions(transaction_id)
);
